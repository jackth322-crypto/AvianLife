    async function initDatabase() {
      const SQL = await initSqlJs({
        locateFile: file => `https://cdnjs.cloudflare.com/ajax/libs/sql.js/1.8.0/${file}`
      });

      sqliteDB = new SQL.Database();
      sqliteDB.run(`
        CREATE TABLE sightings (
          id TEXT PRIMARY KEY,
          species_code TEXT,
          date TEXT,
          lat REAL,
          lng REAL,
          location_name TEXT,
          state TEXT,
          country TEXT,
          notes TEXT,
          photo_url TEXT,
          is_cover INTEGER DEFAULT 0
        );
      `);

      const stmt = sqliteDB.prepare("INSERT INTO sightings VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)");
      SEED_SIGHTINGS.forEach(row => stmt.run(row));
      stmt.free();

      // Load Scraped Truth Database from avian_ebird_data.json
      try {
        const res = await fetch('./avian_ebird_data.json');
        if (res.ok) {
          DATABASE_TRUTH = await res.json();
          document.getElementById('data-status-badge').innerText = 'Truth JSON Loaded';
        }
      } catch (e) {
        console.warn("avian_ebird_data.json not loaded yet, using baseline memory fallback");
      }
    }

    function querySQL(sql, params = []) {
      if (!sqliteDB) return [];
      const stmt = sqliteDB.prepare(sql);
      stmt.bind(params);
      const results = [];
      while (stmt.step()) results.push(stmt.getAsObject());
      stmt.free();
      return results;
    }

    function executeSQL(sql, params = []) {
      if (!sqliteDB) return;
      sqliteDB.run(sql, params);
    }

        ? "SELECT COUNT(DISTINCT species_code) as cnt FROM sightings WHERE country = 'United States'"
        : "SELECT COUNT(DISTINCT species_code) as cnt FROM sightings WHERE LOWER(state) = LOWER(?)";
      const params = isCountry ? [] : [activeState];

      const count = loggedRes[0]?.cnt || 0;

    function generateSparklineHTML(monthly) {
      if (!monthly || monthly.length !== 12) return '';
      return monthly.map((val, idx) => {
        const height = Math.max(3, Math.min(14, Math.round(val * 16)));
        const isActive = val > 0.05 ? 'active' : '';
        const isCurrent = idx === CURRENT_MONTH ? 'current-month' : '';
        return `<span class="sparkline-bar ${isActive} ${isCurrent}" style="height: ${height}px;"></span>`;
      }).join('');
    }

    function getSeasonalityBadge(status) {
      if (status === 'Year-round Resident') return '<span class="text-[9px] bg-emerald-950 text-emerald-400 border border-emerald-800 px-1.5 py-0.2 rounded font-medium">Year-round</span>';
      if (status === 'Summer Breeding') return '<span class="text-[9px] bg-amber-950 text-amber-400 border border-amber-800 px-1.5 py-0.2 rounded font-medium">Summer Breeding</span>';
      if (status === 'Winter Visitor') return '<span class="text-[9px] bg-blue-950 text-blue-400 border border-blue-800 px-1.5 py-0.2 rounded font-medium">Winter Visitor</span>';

      return '<span class="text-[9px] bg-stone-800 text-stone-400 border border-stone-700 px-1.5 py-0.2 rounded font-medium">Seasonal</span>';
    }

    function renderMapMarkers() {
      if (!markersLayer || !sqliteDB) return;
      markersLayer.clearLayers();

      const currentZoom = mapInstance.getZoom();
      const showPhotoChits = currentZoom >= 11;

      let sql = "SELECT * FROM sightings WHERE 1=1";
      const params = [];
      if (currentSpeciesFilter !== 'all') {
        sql += " AND species_code = ?";
        params.push(currentSpeciesFilter);
      }

      const allSightings = querySQL(sql, params);
      const now = new Date();

      const filtered = allSightings.filter(s => {
        if (currentTimeFilter === 'all') return true;
        const diffDays = (now - new Date(s.date)) / (1000 * 60 * 60 * 24);
        if (currentTimeFilter === '7d') return diffDays <= 7;
        if (currentTimeFilter === '30d') return diffDays <= 30;
        if (currentTimeFilter === '1y') return diffDays <= 365;
        return true;
      });

      filtered.forEach(s => {
        const tax = (DATABASE_TRUTH.taxonomy && DATABASE_TRUTH.taxonomy[s.species_code]) || { comName: s.species_code.toUpperCase() };
        let marker;
        if (showPhotoChits && s.photo_url) {
          const chitIcon = L.divIcon({
            className: 'bird-chit-icon',
            html: '',
            iconSize: [40, 40],
            iconAnchor: [20, 20]
          });
          marker = L.marker([s.lat, s.lng], { icon: chitIcon });
          marker.on('add', () => {
            const el = marker.getElement();
            if (el) el.style.backgroundImage = `url('${s.photo_url}')`;
          });
        } else {
          marker = L.circleMarker([s.lat, s.lng], {
            radius: 8,
            fillColor: '#059669',
            color: '#ffffff',
            weight: 2,
            fillOpacity: 0.95
          });
        }

        const popup = `
          <div style="min-width: 170px; font-family: sans-serif; color: #f5f5f4; padding: 6px;">
            ${s.photo_url ? `<img src="${s.photo_url}" style="width: 100%; aspect-ratio: 1/1; object-fit: cover; border-radius: 8px; margin-bottom: 6px;" />` : ''}
            <div style="font-weight: 700; font-size: 13px;">${tax.comName}</div>
            <div style="font-size: 10px; color: #a8a29e; margin-top: 3px;">📍 ${s.location_name}<br>📅 ${s.date}</div>
            ${s.notes ? `<div style="font-size: 10px; color: #d6d3d1; margin-top: 4px; font-style: italic;">"${s.notes}"</div>` : ''}
          </div>
        `;
        marker.bindPopup(popup);
        markersLayer.addLayer(marker);
      });
    }

    function renderLifeList() {

      const sql = `
        SELECT species_code, MIN(date) as first_logged, MAX(date) as last_logged,
               COUNT(id) as log_count,
               COALESCE(MAX(CASE WHEN is_cover = 1 THEN photo_url END), MAX(photo_url)) as cover_photo,
               MAX(location_name) as top_location

    function renderRadar() {
      const stateCode = STATE_CODES[activeState] || 'US-NC';
      const stateEntry = DATABASE_TRUTH.states ? DATABASE_TRUTH.states[stateCode] : null;
      const logged = querySQL("SELECT DISTINCT species_code FROM sightings");
      const loggedCodes = new Set(logged.map(l => l.species_code));

      const targets = [];
      const speciesRoster = stateEntry ? stateEntry.species : [];
      speciesRoster.forEach(code => {
        if (!loggedCodes.has(code)) {
          const tax = (DATABASE_TRUTH.taxonomy && DATABASE_TRUTH.taxonomy[code]) || { comName: code.toUpperCase(), sciName: 'Aves sp.' };
          const season = (stateEntry && stateEntry.seasonality && stateEntry.seasonality[code]) || { status: 'Seasonal', monthly: [] };
          const prob = season.monthly ? (season.monthly[CURRENT_MONTH] || 0.05) : 0.05;
          targets.push({
            code,
            comName: tax.comName,
            sciName: tax.sciName,
            status: season.status,
            prob: prob,
            monthly: season.monthly
          });
        }
      });

      targets.sort((a, b) => b.prob - a.prob);

      const container = document.getElementById('radar-list');
      if (targets.length === 0) {
        container.innerHTML = '<div class="text-center py-8 text-stone-500">All regional targets logged!</div>';
        return;
      }

      container.innerHTML = targets.slice(0, 15).map(t => `
        <div class="bg-stone-900 border border-stone-800 p-3.5 rounded-xl flex items-center justify-between">
          <div>
            <div class="flex items-center gap-1.5">
              <h4 class="font-bold text-xs">${t.comName}</h4>
              ${getSeasonalityBadge(t.status)}
            </div>
            <div class="text-[10px] text-stone-400 italic">${t.sciName}</div>
            <div class="flex items-center gap-2 mt-1.5">
              <div class="flex items-end h-3">${generateSparklineHTML(t.monthly)}</div>
              <span class="text-[10px] text-emerald-400 font-mono">${(t.prob * 100).toFixed(0)}% frequency now</span>
            </div>
          </div>
          <button onclick="openUploadModal(); document.getElementById('form-species').value='${t.code}';" class="bg-stone-800 text-stone-200 text-xs px-2.5 py-1.5 rounded-lg border border-stone-700 font-medium">Log</button>
        </div>
      `).join('');

      lucide.createIcons();
    }

    // --- CROPPER ENGINE ---
    function handleImageSelection(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (event) => {
        originalImageObj.onload = () => {
          document.getElementById('cropper-container').classList.remove('hidden');
          const img = document.getElementById('crop-image');
          img.src = originalImageObj.src;
          resetCrop();
        };
        originalImageObj.src = event.target.result;
      };
      reader.readAsDataURL(file);
    }

    function resetCrop() {
      cropState.scale = 1; cropState.x = 0; cropState.y = 0;
      document.getElementById('crop-zoom').value = 1;
      const viewport = document.getElementById('crop-viewport');
      const img = document.getElementById('crop-image');
      const vWidth = viewport.clientWidth, vHeight = viewport.clientHeight;
      if (originalImageObj.width / originalImageObj.height > 1) {
        img.style.height = `${vHeight}px`; img.style.width = 'auto';
      } else {
        img.style.width = `${vWidth}px`; img.style.height = 'auto';
      }
      updateCropTransform();
    }

    function updateCropTransform() {
      cropState.scale = parseFloat(document.getElementById('crop-zoom').value);
      const img = document.getElementById('crop-image');
      const viewport = document.getElementById('crop-viewport');
      const offsetX = (viewport.clientWidth - img.offsetWidth) / 2;
      const offsetY = (viewport.clientHeight - img.offsetHeight) / 2;
      img.style.transform = `translate(${offsetX + cropState.x}px, ${offsetY + cropState.y}px) scale(${cropState.scale})`;
    }

    const cropViewport = document.getElementById('crop-viewport');
    function startDrag(clientX, clientY) { cropState.isDragging = true; cropState.startX = clientX - cropState.x; cropState.startY = clientY - cropState.y; }
    function moveDrag(clientX, clientY) { if (!cropState.isDragging) return; cropState.x = clientX - cropState.startX; cropState.y = clientY - cropState.startY; updateCropTransform(); }
    function endDrag() { cropState.isDragging = false; }

    cropViewport.addEventListener('mousedown', (e) => startDrag(e.clientX, e.clientY));
    window.addEventListener('mousemove', (e) => moveDrag(e.clientX, e.clientY));
    window.addEventListener('mouseup', endDrag);
    cropViewport.addEventListener('touchstart', (e) => { if (e.touches.length === 1) startDrag(e.touches[0].clientX, e.touches[0].clientY); });
    window.addEventListener('touchmove', (e) => { if (e.touches.length === 1) moveDrag(e.touches[0].clientX, e.touches[0].clientY); });
    window.addEventListener('touchend', endDrag);

    function generateCroppedDataUrl() {
      if (!originalImageObj.src || !document.getElementById('crop-image').src) return null;
      const canvas = document.createElement('canvas');
      const size = 700;
      canvas.width = size; canvas.height = size;
      const ctx = canvas.getContext('2d');
      const viewport = document.getElementById('crop-viewport');
      const img = document.getElementById('crop-image');
      const vSize = viewport.clientWidth, ratio = size / vSize;

      const renderedWidth = img.offsetWidth * cropState.scale;
      const renderedHeight = img.offsetHeight * cropState.scale;
      const renderedLeft = ((vSize - img.offsetWidth) / 2 + cropState.x) - ((renderedWidth - img.offsetWidth) / 2);
      const renderedTop = ((vSize - img.offsetHeight) / 2 + cropState.y) - ((renderedHeight - img.offsetHeight) / 2);

      ctx.fillStyle = '#0c0a09'; ctx.fillRect(0, 0, size, size);
      ctx.drawImage(originalImageObj, renderedLeft * ratio, renderedTop * ratio, renderedWidth * ratio, renderedHeight * ratio);
      return canvas.toDataURL('image/jpeg', 0.9);
    }

    function fetchGPSLocation() {
      const btn = document.getElementById('btn-gps');
      btn.innerHTML = '<span class="animate-spin text-xs">🌀</span> Locating...';
      navigator.geolocation.getCurrentPosition(async (pos) => {
        const { latitude, longitude } = pos.coords;
        setFormCoordinates(latitude, longitude);
        try {
          const res = await fetch(`https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${latitude}&longitude=${longitude}&localityLanguage=en`);
          const data = await res.json();
          if (data && data.principalSubdivision) {
            document.getElementById('form-state').value = data.principalSubdivision;
            document.getElementById('form-location').value = data.locality || data.city || 'Field Sighting';
          }
        } catch (e) {}
        btn.innerHTML = '<i data-lucide="check" class="w-3.5 h-3.5 text-emerald-400"></i> GPS Locked';
        lucide.createIcons();
      }, (err) => {
        btn.innerHTML = '<i data-lucide="navigation" class="w-3.5 h-3.5 text-emerald-400"></i> Use My GPS';
        lucide.createIcons();
      }, { enableHighAccuracy: true });
    }

    function setFormCoordinates(lat, lng) {
      selectedCoords = { lat, lng };
      if (pickerMarker) pickerMarker.setLatLng([lat, lng]);
      if (pickerMap) pickerMap.setView([lat, lng], 13);
    }

    function togglePickerMap() {
      const drawer = document.getElementById('picker-map-drawer');
      drawer.classList.toggle('hidden');
      if (!pickerMap) {
        pickerMap = L.map('picker-map', { zoomControl: false }).setView([selectedCoords.lat, selectedCoords.lng], 12);
        L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 18 }).addTo(pickerMap);
        pickerMarker = L.marker([selectedCoords.lat, selectedCoords.lng], { draggable: true }).addTo(pickerMap);
        pickerMarker.on('dragend', (e) => setFormCoordinates(e.target.getLatLng().lat, e.target.getLatLng().lng));
        pickerMap.on('click', (e) => setFormCoordinates(e.latlng.lat, e.latlng.lng));
      }
      setTimeout(() => pickerMap.invalidateSize(), 150);
    }

    function saveSightingRecord() {
      const speciesCode = document.getElementById('form-species').value;
      const location = document.getElementById('form-location').value || 'Field Observation';
      const date = document.getElementById('form-date').value;
      const state = document.getElementById('form-state').value || activeState;
      const notes = document.getElementById('form-notes').value;
      const isCover = document.getElementById('form-is-cover').checked ? 1 : 0;
      let photoUrl = generateCroppedDataUrl() || 'https://images.unsplash.com/photo-1548802673-380ab8ebc7b7?auto=format&fit=crop&w=800&q=85';

      const id = 's_' + Date.now();
      if (isCover === 1) executeSQL("UPDATE sightings SET is_cover = 0 WHERE species_code = ?", [speciesCode]);
      executeSQL("INSERT INTO sightings VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)", [id, speciesCode, date, selectedCoords.lat, selectedCoords.lng, location, state, 'United States', notes, photoUrl, isCover]);

      closeUploadModal();
      renderMapMarkers();
      updateRegionalHUD();
      renderLifeList();
      renderGallery();
      populateFilterDropdowns();
      renderRadar();
    }

    function switchTab(tab) {
      document.getElementById('view-map').classList.toggle('hidden', tab !== 'map');
      document.getElementById('view-lifelist').classList.toggle('hidden', tab !== 'lifelist');
      document.getElementById('view-gallery').classList.toggle('hidden', tab !== 'gallery');
      document.getElementById('view-radar').classList.toggle('hidden', tab !== 'radar');

      ['map', 'lifelist', 'gallery', 'radar'].forEach(t => {
        document.getElementById('nav-btn-' + t).className = t === tab ? 'flex flex-col items-center gap-1 text-emerald-400' : 'flex flex-col items-center gap-1 text-stone-400';
      });

      if (tab === 'map' && mapInstance) setTimeout(() => mapInstance.invalidateSize(), 200);
      if (tab === 'radar') renderRadar();
    }

        uploadSelect.innerHTML += `<option value="${code}">${tax.comName} (${tax.sciName})</option>`;
      });
    }
